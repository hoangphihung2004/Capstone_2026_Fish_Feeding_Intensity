# ------------------------------------------------------------------------------
# 1. TOKEN HUGGING FACE (token Write, chỉ cần làm 1 lần trên máy)
#    Không dán token vào file này. Lưu token vào /marimo/token.txt:
#    echo 'hf_xxx' > /marimo/token.txt
# ------------------------------------------------------------------------------
export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' /marimo/token.txt | head -n 1)
if [ -z "$HF_TOKEN" ]; then echo "⚠️ Chưa có token trong /marimo/token.txt"; fi
hf auth login --token "$HF_TOKEN" || huggingface-cli login --token "$HF_TOKEN"

# ------------------------------------------------------------------------------
# 2. TẢI DATASET (bỏ qua nếu /marimo/Fish_Feeding_Intensity_Dataset đã có)
# ------------------------------------------------------------------------------
sudo apt update && sudo apt install git-lfs -y
git lfs install
git config --global lfs.concurrenttransfers 64
cd /marimo && nohup git clone https://huggingface.co/datasets/manhmitcf/Fish_Feeding_Intensity_Dataset > /marimo/clone.log 2>&1 &
tail -n +1 -f /marimo/clone.log

mv /marimo/Fish_Feeding_Intensity_Dataset/audio/audio/* /marimo/Fish_Feeding_Intensity_Dataset/audio/
mv /marimo/Fish_Feeding_Intensity_Dataset/video/video/* /marimo/Fish_Feeding_Intensity_Dataset/video/
rm -rf /marimo/Fish_Feeding_Intensity_Dataset/audio/audio
rm -rf /marimo/Fish_Feeding_Intensity_Dataset/video/video
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.mp4" | wc -l
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.wav" | wc -l

# ------------------------------------------------------------------------------
# 3. LẤY CODE VÀ CÀI THƯ VIỆN
# ------------------------------------------------------------------------------
cd /marimo
git clone https://github.com/hoangphihung2004/Capstone_2026_Fish_Feeding_Intensity.git
cd /marimo/Capstone_2026_Fish_Feeding_Intensity
git fetch origin
git checkout U_FFIA27K_audio_robustness
git pull
cd U_FFIA27K_audio
pip install --upgrade pip
pip install -r requirements.txt
# Cài torchaudio cùng phiên bản với torch (máy marimo: torch 2.11.0). Torch khác thì đổi số 2.11.0 cho khớp.
pip install --no-deps --force-reinstall "torchaudio==2.11.0"
# Kiểm tra: phải in ra torch 2.11.0 | torchaudio 2.11.0
python -c "import torch, torchaudio; print('torch', torch.__version__, '| torchaudio', torchaudio.__version__)"

# ------------------------------------------------------------------------------
# 4. KIỂM TRA WEIGHT fold 0-4 có trên máy (đường dẫn trong config/robustness_paths.json)
#    Phải in ra 5 dòng OK. Có dòng MISSING thì tải weight về /marimo/Results trước.
# ------------------------------------------------------------------------------
python -c "import json,os;[print('OK     ' if os.path.isfile(p) else 'MISSING', f, p) for f,p in json.load(open('config/robustness_paths.json'))['checkpoints'].items()]"

# ------------------------------------------------------------------------------
# 5. CHẠY: sanity ở fold 0, qua thì tự chạy TOÀN BỘ 5 fold (0,1,2,3,4). Mỗi fold xong tự upload.
#    Không git checkout nhánh khác trong cùng thư mục khi lệnh này đang chạy.
# ------------------------------------------------------------------------------
cd /marimo/Capstone_2026_Fish_Feeding_Intensity && git fetch origin && git checkout U_FFIA27K_audio_robustness && git pull && cd U_FFIA27K_audio
# Cài torchaudio cùng phiên bản với torch (máy marimo: torch 2.11.0), tránh lỗi libtorchaudio.so: undefined symbol
pip install --no-deps --force-reinstall "torchaudio==2.11.0"
python -c "import torch, torchaudio; print('torch', torch.__version__, '| torchaudio', torchaudio.__version__)"
nohup sh -c "python evaluate_robustness.py --folds 0 --sanity && python evaluate_robustness.py --folds 0 1 2 3 4" > robustness.log 2>&1 &
tail -n +1 -f robustness.log

# Đọc log:
#   - Sanity qua: các dòng [sanity] ... PASS, rồi chạy fold 0-4; mỗi fold xong có dòng "Uploaded ..._fold0X_full_....zip"
#   - Sanity FAIL: dừng với "Sanity checks failed on folds: [0]" -> gửi robustness.log hoặc outputs/robustness/*/fold_00/sanity.json

# Nếu bị ngắt giữa chừng: chạy tiếp các mức chưa xong
nohup python evaluate_robustness.py --resume > robustness.log 2>&1 &

# Nếu upload lỗi (log in ra fold nào lỗi): upload lại mà không chạy lại
python evaluate_robustness.py --upload-only --folds 0 1 2 3 4
