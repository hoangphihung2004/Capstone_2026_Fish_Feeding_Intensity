# ------------------------------------------------------------------------------
# 4. KIỂM TRA WEIGHT fold 0-4 có trên máy (đường dẫn trong config/robustness_paths.json)
#    Phải in ra 5 dòng OK. Có dòng MISSING thì tải weight về /marimo/Results trước.
# ------------------------------------------------------------------------------
python -c "import json,os;[print('OK     ' if os.path.isfile(p) else 'MISSING', f, p) for f,p in json.load(open('config/robustness_paths.json'))['checkpoints'].items()]"

# ------------------------------------------------------------------------------
# 5. CHẠY (mỗi fold xong tự upload). CHỌN 1 TRONG 3 CÁCH:
# ------------------------------------------------------------------------------
# Cách A (khuyên dùng): kiểm tra sanity ở fold 0, qua thì tự chạy TOÀN BỘ 5 fold (0,1,2,3,4)
nohup sh -c "python evaluate_robustness.py --folds 0 --sanity && python evaluate_robustness.py --folds 0 1 2 3 4" > robustness.log 2>&1 &

# Cách B: chạy luôn TOÀN BỘ 5 fold, bỏ qua sanity
nohup python evaluate_robustness.py --folds 0 1 2 3 4 > robustness.log 2>&1 &

# Cách C: chạy riêng 1 fold (đổi số 0 thành fold muốn chạy)
nohup python evaluate_robustness.py --folds 0 > robustness.log 2>&1 &

# Xem log
tail -n +1 -f robustness.log

# Nếu bị ngắt giữa chừng: chạy tiếp các mức chưa xong
nohup python evaluate_robustness.py --resume > robustness.log 2>&1 &

# Nếu upload lỗi (log in ra fold nào lỗi): upload lại mà không chạy lại
python evaluate_robustness.py --upload-only --folds 0 1 2 3 4
