# 5. CHẠY: sanity ở fold 0, qua thì tự chạy TOÀN BỘ 5 fold (0,1,2,3,4). Mỗi fold xong tự upload.

TV=$(python -c "import torch;print(torch.__version__)"); V=${TV%%+*}; if [[ $TV == *+* ]]; then pip install --no-deps --force-reinstall "torchaudio==$V" --index-url https://download.pytorch.org/whl/${TV#*+}; else pip install --no-deps --force-reinstall "torchaudio==$V"; fi
